# Prerequisite

```
sudo apt update

sudo apt install -y python-is-python3 python3.12-venv

curl -fsSL https://deb.nodesource.com/setup_20.x | sudo -E bash -

sudo apt install nodejs -y

python -m venv .venv

source .venv/bin/activate

pip install -e .

cd ..

pip install -e .

cd ui/frontend

sudo npm install -g gatsby-cli

sudo npm install -g yarn

yarn install

export NODE_OPTIONS="--no-warnings"

yarn build

chmod +x run.sh

chmod +x run-github.sh
```

# Configure to run on github
- Open run-github.com

- Edit this line, `export CODESPACE_ID="sturdy-robot-jwv4gq46j9gfpp7q"` to your codespace ID.

- Forward port 8000, 3000, 5000

# Run Web UI

```
cd frontend

./run-github.sh 
```
