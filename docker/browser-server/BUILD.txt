# Prerequisite

```
sudo apt-get update

sudo apt-get install -y \
    git \
    xvfb \
    x11vnc \
    openbox \
    supervisor \
    python3 \
    python3-pip \
    novnc \
    websockify \
    procps \
    xdg-utils \
    python3-xdg \
    x11-xserver-utils \
    rsync \
    --no-install-recommends

sudo git clone --depth 1 --branch v1.6.0 https://github.com/novnc/noVNC.git /usr/local/novnc

sudo git clone --depth 1 --branch v0.13.0 https://github.com/novnc/websockify /usr/local/novnc/utils/websockify

source ../ui/.venv # You need to build the web server first

sudo npm install .

sudo npx playwright@1.55 install --with-deps chromium

chmod +x run.sh

chmod +x launch-session.sh
```

# Run Browser Server

```
./run.sh
```
